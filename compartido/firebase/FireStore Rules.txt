rules_version = '2';

// Academia Gloria Valentina
// Identidad Multi-Persona · Fase 1.9
// Login funcional separado del UID Firebase
// Administración de Usuarios + acceso funcional por Persona Activa · Spark / sin Cloud Functions

service cloud.firestore {
  match /databases/{database}/documents {

    function estaAutenticado() {
      return request.auth != null;
    }

    function esPropietario(userId) {
      return estaAutenticado()
        && request.auth.uid == userId;
    }

    function existeUsuarioActual() {
      return estaAutenticado()
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function personaIdUsuarioActual() {
      return get(
        /databases/$(database)/documents/usuarios/$(request.auth.uid)
      ).data.personaId;
    }

    function rutaRelacionCon(personaId) {
      return /databases/$(database)/documents/personaRelaciones/$(personaIdUsuarioActual() + "__" + personaId);
    }

    function tieneRelacionActivaCon(personaId) {
      let relacionPath = rutaRelacionCon(personaId);

      return existeUsuarioActual()
        && personaId is string
        && exists(relacionPath)
        && get(relacionPath).data.activo == true;
    }

    function tieneRolActualActivo() {
      let asignacionPath =
        /databases/$(database)/documents/usuarioRoles/$(request.auth.uid);

      return estaAutenticado()
        && exists(asignacionPath)
        && get(asignacionPath).data.activo == true
        && get(asignacionPath).data.roleId is string
        && exists(
          /databases/$(database)/documents/roles/$(get(asignacionPath).data.roleId)
        )
        && get(
          /databases/$(database)/documents/roles/$(get(asignacionPath).data.roleId)
        ).data.activo == true;
    }

    function nivelRolActual() {
      let asignacionPath =
        /databases/$(database)/documents/usuarioRoles/$(request.auth.uid);

      return get(
        /databases/$(database)/documents/roles/$(get(asignacionPath).data.roleId)
      ).data.nivelAcceso;
    }

    function puedeConsultarPersona(personaId) {
      return tieneRelacionActivaCon(personaId)
        && tieneRolActualActivo();
    }

    function puedeGestionarPersona(personaId) {
      let relacionPath = rutaRelacionCon(personaId);

      return tieneRelacionActivaCon(personaId)
        && tieneRolActualActivo()
        && get(relacionPath).data.nivelAcceso == "gestion"
        && (
          nivelRolActual() == "gestion"
          || nivelRolActual() == "administracion"
        );
    }

    function personaIdUsuarioObjetivo(userId) {
      return get(
        /databases/$(database)/documents/usuarios/$(userId)
      ).data.personaId;
    }

    // ADMINISTRACIÓN · Fase 1.7 / 1.8
    //
    // Un administrador se determina exclusivamente mediante el modelo
    // vigente USER_ROLE -> ROLE.
    //
    // Requisitos:
    // - usuario autenticado;
    // - USER_ROLE existente y activo;
    // - ROLE existente y activo;
    // - ROLE.nivelAcceso == "administracion".
    function esAdministrador() {
      let asignacionPath =
        /databases/$(database)/documents/usuarioRoles/$(request.auth.uid);

      let asignacion = get(asignacionPath);

      return estaAutenticado()
        && exists(asignacionPath)
        && asignacion.data.activo == true
        && asignacion.data.roleId is string
        && exists(
          /databases/$(database)/documents/roles/$(asignacion.data.roleId)
        )
        && get(
          /databases/$(database)/documents/roles/$(asignacion.data.roleId)
        ).data.activo == true
        && get(
          /databases/$(database)/documents/roles/$(asignacion.data.roleId)
        ).data.nivelAcceso == "administracion";
    }

    // LOGIN FUNCIONAL · Fase 1.6 / 1.8
    //
    // El usuario aún NO está autenticado cuando escribe su login.
    // Se permite únicamente GET directo a accesosLogin/{login}.
    //
    // LIST permanece bloqueado para usuarios normales.
    // Un administrador autenticado puede consultar y mantener la colección
    // para la Gestión de Usuarios.
    match /accesosLogin/{loginId} {
      allow get: if true;
      allow list: if esAdministrador();

      allow create, update, delete: if esAdministrador();
    }

    // USER + SUBCOLECCIONES FUNCIONALES · Fase 1.8
    //
    // - Propietario y Administración conservan su comportamiento existente.
    // - Un Usuario relacionado puede leer el USER mínimo de la Persona objetivo
    //   para resolver personaId -> userId.
    // - El acceso relacionado se abre únicamente a las subcolecciones
    //   funcionales declaradas explícitamente debajo.
    // - Escritura relacionada exige acceso efectivo de gestión salvo contrato
    //   específico documentado, como la Bitácora de Acompañamiento.
    // - El resto de subcolecciones continúa restringido al propietario/admin.
    match /usuarios/{userId} {
      allow read: if esPropietario(userId)
        || esAdministrador()
        || (
          resource.data.personaId is string
          && puedeConsultarPersona(resource.data.personaId)
        );

      allow create, update, delete: if esPropietario(userId) || esAdministrador();

      match /eventos/{eventoId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // BAÚL DEL ALUMNO · Persona Activa
      //
      // Mismo contrato funcional que Calendario:
      // - propietario/admin/relación de consulta pueden leer;
      // - propietario/admin/relación con gestión pueden modificar.
      match /baul/{elementoId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // Los adjuntos del Baúl se separan para que listar registros no cargue
      // documentos. La interfaz limita V1 a 600 KB; la regla añade además una
      // barrera física sobre el Data URL persistido.
      match /baulAdjuntos/{elementoId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update: if (
            esPropietario(userId)
            || esAdministrador()
            || puedeGestionarPersona(personaIdUsuarioObjetivo(userId))
          )
          && request.resource.data.dataUrl is string
          && request.resource.data.dataUrl.size() <= 900000;

        allow delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // BITÁCORA DE ACOMPAÑAMIENTO · V1
      //
      // Contrato específico de colaboración:
      // - el alumno/propietario solo lee entradas expresamente compartidas con él;
      // - una relación activa con nivel efectivo de consulta puede leer las
      //   entradas para adultos/profesionales y sus propias entradas privadas;
      // - familia/profesionales relacionados pueden crear únicamente entradas
      //   cuya autoría coincide con el USER autenticado;
      // - una entrada publicada es inmutable en V1 salvo por una única respuesta
      //   estructurada registrada por otra persona autorizada;
      // - no hay delete en V1.
      match /bitacoraAcompanamiento/{entradaId} {
        allow read: if esAdministrador()
          || (
            esPropietario(userId)
            && resource.data.visibleParaPersonaActiva == true
          )
          || (
            puedeConsultarPersona(personaIdUsuarioObjetivo(userId))
            && (
              resource.data.createdBy == request.auth.uid
              || resource.data.visibilidad == "adultos-profesionales"
              || resource.data.visibilidad == "todos-relacionados"
            )
          );

        allow create: if (
            esAdministrador()
            || puedeConsultarPersona(personaIdUsuarioObjetivo(userId))
          )
          && request.resource.data.keys().hasOnly([
            "schemaVersion",
            "tipo",
            "tipoOtros",
            "titulo",
            "mensaje",
            "destino",
            "destinoOtros",
            "visibilidad",
            "visibilidadOtros",
            "visibleParaPersonaActiva",
            "requiereRespuesta",
            "estado",
            "personaId",
            "createdAt",
            "createdBy",
            "createdByPersonaId",
            "createdByNombre",
            "createdByRol",
            "createdByRelacion",
            "updatedAt",
            "updatedBy",
            "updatedByNombre",
            "respuestaTexto",
            "respuestaCreatedAt",
            "respuestaCreatedBy",
            "respuestaCreatedByPersonaId",
            "respuestaCreatedByNombre",
            "respuestaCreatedByRol"
          ])
          && request.resource.data.schemaVersion == 1
          && request.resource.data.personaId == personaIdUsuarioObjetivo(userId)
          && request.resource.data.createdBy == request.auth.uid
          && request.resource.data.createdByPersonaId == personaIdUsuarioActual()
          && request.resource.data.createdByNombre is string
          && request.resource.data.createdByNombre.size() > 0
          && request.resource.data.createdByNombre.size() <= 180
          && request.resource.data.createdByRol is string
          && request.resource.data.createdByRol.size() <= 120
          && request.resource.data.createdByRelacion is string
          && request.resource.data.createdByRelacion.size() <= 120
          && request.resource.data.createdAt == request.time
          && request.resource.data.updatedAt == request.time
          && request.resource.data.updatedBy == request.auth.uid
          && request.resource.data.updatedByNombre == request.resource.data.createdByNombre
          && request.resource.data.titulo is string
          && request.resource.data.titulo.size() > 0
          && request.resource.data.titulo.size() <= 180
          && request.resource.data.mensaje is string
          && request.resource.data.mensaje.size() > 0
          && request.resource.data.mensaje.size() <= 6000
          && request.resource.data.tipo in [
            "observacion",
            "recomendacion",
            "duda",
            "sugerencia",
            "acuerdo",
            "seguimiento",
            "mensaje",
            "asunto-revisar",
            "otros"
          ]
          && request.resource.data.tipoOtros is string
          && request.resource.data.tipoOtros.size() <= 180
          && (
            (request.resource.data.tipo == "otros"
              && request.resource.data.tipoOtros.size() > 0)
            || (request.resource.data.tipo != "otros"
              && request.resource.data.tipoOtros == "")
          )
          && request.resource.data.destino in [
            "general",
            "alumno",
            "familia",
            "profesionales",
            "familia-profesionales",
            "otros"
          ]
          && request.resource.data.destinoOtros is string
          && request.resource.data.destinoOtros.size() <= 180
          && (
            (request.resource.data.destino == "otros"
              && request.resource.data.destinoOtros.size() > 0)
            || (request.resource.data.destino != "otros"
              && request.resource.data.destinoOtros == "")
          )
          && request.resource.data.visibilidad in [
            "solo-autor",
            "adultos-profesionales",
            "todos-relacionados",
            "otros"
          ]
          && request.resource.data.visibilidadOtros is string
          && request.resource.data.visibilidadOtros.size() <= 180
          && (
            (request.resource.data.visibilidad == "otros"
              && request.resource.data.visibilidadOtros.size() > 0)
            || (request.resource.data.visibilidad != "otros"
              && request.resource.data.visibilidadOtros == "")
          )
          && request.resource.data.visibleParaPersonaActiva
            == (request.resource.data.visibilidad == "todos-relacionados")
          && request.resource.data.requiereRespuesta is bool
          && (
            request.resource.data.requiereRespuesta == false
            || request.resource.data.visibilidad in [
              "adultos-profesionales",
              "todos-relacionados"
            ]
          )
          && request.resource.data.estado == "abierta"
          && request.resource.data.respuestaTexto == ""
          && request.resource.data.respuestaCreatedAt == null
          && request.resource.data.respuestaCreatedBy == ""
          && request.resource.data.respuestaCreatedByPersonaId == ""
          && request.resource.data.respuestaCreatedByNombre == ""
          && request.resource.data.respuestaCreatedByRol == "";

        allow update: if (
            esAdministrador()
            || (
              puedeConsultarPersona(personaIdUsuarioObjetivo(userId))
              && resource.data.visibilidad in [
                "adultos-profesionales",
                "todos-relacionados"
              ]
            )
          )
          && resource.data.requiereRespuesta == true
          && resource.data.respuestaTexto == ""
          && resource.data.createdBy != request.auth.uid
          && request.resource.data.personaId == resource.data.personaId
          && request.resource.data.createdBy == resource.data.createdBy
          && request.resource.data.respuestaTexto is string
          && request.resource.data.respuestaTexto.size() > 0
          && request.resource.data.respuestaTexto.size() <= 4000
          && request.resource.data.respuestaCreatedBy == request.auth.uid
          && request.resource.data.respuestaCreatedByPersonaId == personaIdUsuarioActual()
          && request.resource.data.respuestaCreatedByNombre is string
          && request.resource.data.respuestaCreatedByNombre.size() > 0
          && request.resource.data.respuestaCreatedByNombre.size() <= 180
          && request.resource.data.respuestaCreatedByRol is string
          && request.resource.data.respuestaCreatedByRol.size() <= 120
          && request.resource.data.respuestaCreatedAt == request.time
          && request.resource.data.estado == "atendida"
          && request.resource.data.updatedAt == request.time
          && request.resource.data.updatedBy == request.auth.uid
          && request.resource.data.updatedByNombre
            == request.resource.data.respuestaCreatedByNombre
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            "respuestaTexto",
            "respuestaCreatedAt",
            "respuestaCreatedBy",
            "respuestaCreatedByPersonaId",
            "respuestaCreatedByNombre",
            "respuestaCreatedByRol",
            "estado",
            "updatedAt",
            "updatedBy",
            "updatedByNombre"
          ]);

        allow delete: if false;
      }

      match /tareas/{tareaId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      match /evidencias/{evidenciaId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // MOTIVACIÓN Y RECONOCIMIENTO · Recompensas A1/A2
      //
      // El alumno/propietario puede consultar su historia motivacional,
      // pero la creación/edición humana requiere gestión adulta o admin.
      // Fase B deriva los reconocimientos automáticos desde sesiones reales;
      // no abre una escritura automática del alumno sobre esta colección.
      match /reconocimientos/{reconocimientoId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // MI RINCÓN DE LECTURA · Persona Activa
      //
      // La familia/profesional relacionado puede consultar las sesiones
      // de la Persona Activa. La modificación exige nivel efectivo de gestión.
      match /sesionesLectura/{sesionId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // CRECIENDO POR DENTRO · Sesiones de Semillas · Persona Activa
      //
      // Sigue el mismo contrato funcional que el resto de sesiones:
      // consulta para relaciones activas y modificación solo con gestión.
      match /sesionesSemillas/{sesionId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // SESIONES ACADÉMICAS · Persona Activa
      //
      // Sigue el mismo contrato de acceso funcional que las demás evidencias:
      // familia/profesional relacionado puede consultar el resultado histórico;
      // modificar exige nivel efectivo de gestión.
      match /sesionesAcademicas/{sesionId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
      }

      // DETECTIVES DE PROBLEMAS · Persona Activa
      //
      // La familia/profesional relacionado puede consultar el progreso y las
      // resoluciones de la Persona Activa. La modificación exige gestión.
      match /detectivesHistorias/{historiaId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

        allow create, update, delete: if esPropietario(userId)
          || esAdministrador()
          || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));

        match /sesiones/{sesionId} {
          allow read: if esPropietario(userId)
            || esAdministrador()
            || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));

          allow create, update, delete: if esPropietario(userId)
            || esAdministrador()
            || puedeGestionarPersona(personaIdUsuarioObjetivo(userId));
        }
      }

      // BIBLIOTECA ENCANTADA · consulta relacionada para Trabajo realizado
      //
      // Biblioteca y sus audios continúan usando las colecciones legacy
      // usuarios/{uid}/biblioteca y usuarios/{uid}/bibliotecaAudios.
      // Se amplía únicamente la LECTURA a una relación activa con permiso de
      // consulta. La escritura conserva exactamente el comportamiento legacy
      // propietario/admin del fallback inferior.
      match /biblioteca/{libroId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));
      }

      match /bibliotecaAudios/{libroId} {
        allow read: if esPropietario(userId)
          || esAdministrador()
          || puedeConsultarPersona(personaIdUsuarioObjetivo(userId));
      }

      // Compatibilidad legacy de primer nivel.
      //
      // Antes este fallback era recursivo ({documento=**}); por lo tanto también
      // coincidía con reconocimientos/{id} y, como los allow se acumulan,
      // devolvía al propietario permiso genérico de escritura sobre Recompensas.
      // Se limita a una sola subcolección/documento y se excluyen explícitamente
      // las colecciones con contrato de seguridad propio.
      match /{coleccionLegacy}/{documentoLegacy} {
        allow read, write: if !(
            coleccionLegacy in ["reconocimientos", "bitacoraAcompanamiento"]
          )
          && (esPropietario(userId) || esAdministrador());
      }
    }

    // PERSON
    //
    // Comportamiento existente:
    // - el usuario puede consultar su propia Persona;
    // - también una Persona destino cuando existe una relación activa
    //   desde su propia Persona hacia ella.
    //
    // Fase 1.8:
    // - administración puede consultar y mantener Personas.
    match /personas/{personaId} {
      allow read: if esAdministrador()
        || (
          existeUsuarioActual()
          && (
            personaIdUsuarioActual() == personaId
            ||
            (
              exists(
                /databases/$(database)/documents/personaRelaciones/
                $(personaIdUsuarioActual() + "__" + personaId)
              )
              &&
              get(
                /databases/$(database)/documents/personaRelaciones/
                $(personaIdUsuarioActual() + "__" + personaId)
              ).data.activo == true
            )
          )
        );

      allow create, update, delete: if esAdministrador();
    }

    // ROLE
    //
    // Catálogo global de solo lectura para autenticados.
    // La Gestión de Usuarios v0.1 asigna Roles existentes;
    // no administra todavía el catálogo ROLE.
    match /roles/{roleId} {
      allow read: if estaAutenticado();
      allow write: if false;
    }

    // USER_ROLE
    //
    // El ID del documento es el UID Firebase del usuario.
    // - el usuario puede consultar su propia asignación;
    // - administración puede consultar y mantener asignaciones.
    match /usuarioRoles/{userId} {
      allow read: if esPropietario(userId) || esAdministrador();
      allow create, update, delete: if esAdministrador();
    }

    // PERSON_RELATION
    //
    // Comportamiento existente:
    // - un usuario autenticado puede consultar las relaciones
    //   cuyo origen sea su propia PERSON.
    //
    // Fase 1.8:
    // - administración puede consultar y mantener relaciones.
    match /personaRelaciones/{relationId} {
      allow read: if esAdministrador()
        || (
          existeUsuarioActual()
          && resource.data.sourcePersonId == personaIdUsuarioActual()
        );

      allow create, update, delete: if esAdministrador();
    }
  }
}
