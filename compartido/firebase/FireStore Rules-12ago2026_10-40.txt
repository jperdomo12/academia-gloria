service cloud.firestore {
  match /databases/{database}/documents {

    function estaAutenticado() {
      return request.auth != null;
    }

    function esPropietario(userId) {
      return estaAutenticado()
        && request.auth.uid == userId;
    }

    function existeUsuarioActual() {
      return estaAutenticado()
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function personaIdUsuarioActual() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.personaId;
    }

    // LOGIN FUNCIONAL · Fase 1.6
    //
    // El usuario aún NO está autenticado cuando escribe su login.
    // Se permite únicamente GET directo a accesosLogin/{login}.
    //
    // LIST permanece bloqueado para impedir enumerar la colección.
    // Las escrituras desde cliente permanecen completamente cerradas.
    match /accesosLogin/{loginId} {
      allow get: if true;
      allow list: if false;
      allow create, update, delete: if false;
    }

    // Estructura actual: conservar sin cambios funcionales.
    match /usuarios/{userId}/{documento=**} {
      allow read, write: if esPropietario(userId);
    }
   
    // ---------------
    // PERSON:
    // - el usuario puede consultar su propia Persona;
    // - también una Persona destino cuando existe una relación activa
    //   desde su propia Persona hacia ella.
    match /personas/{personaId} {
      allow read: if existeUsuarioActual()
        && (
          personaIdUsuarioActual() == personaId
          ||
          (
            exists(
              /databases/$(database)/documents/personaRelaciones/
              $(personaIdUsuarioActual() + "__" + personaId)
            )
            &&
            get(
              /databases/$(database)/documents/personaRelaciones/
              $(personaIdUsuarioActual() + "__" + personaId)
            ).data.activo == true
          )
        );

      allow write: if false;
    }
    
    // --------------------

    // ROLE: catálogo global de solo lectura para autenticados.
    match /roles/{roleId} {
      allow read: if estaAutenticado();
      allow write: if false;
    }

    // USER_ROLE: el ID del documento es el UID Firebase del usuario.
    match /usuarioRoles/{userId} {
      allow read: if esPropietario(userId);
      allow write: if false;
    }

  	// PERSON_RELATION:
	 // un usuario autenticado puede consultar las relaciones
	 // cuyo origen sea su propia PERSON.
      match /personaRelaciones/{relationId} {
        allow read: if existeUsuarioActual()
          && resource.data.sourcePersonId == personaIdUsuarioActual();
        allow write: if false;
      }
  }
}
